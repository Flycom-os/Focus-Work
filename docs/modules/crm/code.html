📊 1. Воронка продаж DealPipeline.tsx

Воронка продаж похожа на Kanban-доску задач, но имеет бизнес-специфику: внизу каждой колонки (этапа сделки) выводится суммарная стоимость сделок на этом этапе. Это позволяет менеджерам мгновенно оценивать финансовое состояние воронки. 

"use client";

import React, { useState } from "react";
import { DragDropContext, Droppable, Draggable, DropResult } from "@hello-pangea/dnd";

interface Deal {
  id: string;
  title: string;
  amount: number; // Стоимость сделки
  stage: string;
}

const STAGES = [
  { id: "LEAD", name: "Новые лиды" },
  { id: "NEGOTIATION", name: "Переговоры" },
  { id: "CONTRACT", name: "Согласование договора" },
  { id: "PAYMENT", name: "Ожидание оплаты" },
  { id: "COMPLETED", name: "Сделка закрыта" }
];

export default function DealPipeline({ initialDeals }: { initialDeals: Deal[] }) {
  const [deals, setDeals] = useState(initialDeals);

  // Вычисление суммы сделок на конкретном этапе
  const getStageTotalAmount = (stageId: string) => {
    return deals
      .filter(deal => deal.stage === stageId)
      .reduce((sum, deal) => sum + deal.amount, 0);
  };

  const onDragEnd = (result: DropResult) => {
    const { destination, source, draggableId } = result;
    if (!destination) return;
    if (destination.droppableId === source.droppableId) return;

    // Обновляем этап сделки в локальном стейте
    setDeals(prev => {
      return prev.map(deal => {
        if (deal.id === draggableId) {
          return { ...deal, stage: destination.droppableId };
        }
        return deal;
      });
    });

    // Тут вызывается Server Action для записи в БД:
    // updateDealStageAction(draggableId, destination.droppableId);
  };

  return (
    <DragDropContext onDragEnd={onDragEnd}>
      <div class="flex gap-4 p-4 overflow-x-auto select-none">
        {STAGES.map(stage => {
          const stageDeals = deals.filter(d => d.stage === stage.id);
          const totalAmount = getStageTotalAmount(stage.id);

          return (
            <div key={stage.id} class="w-72 bg-slate-50 dark:bg-slate-900 rounded-xl p-4 shrink-0 flex flex-col border border-slate-200 dark:border-slate-800">
              {/* Шапка колонки */}
              <div class="mb-4 border-b pb-2 dark:border-slate-800">
                <h3 class="font-bold text-slate-800 dark:text-slate-200 text-sm">{stage.name}</h3>
                <p class="text-xs text-slate-400 mt-1">
                  Сделок: <span class="font-semibold text-slate-600 dark:text-slate-300">{stageDeals.length}</span>
                </p>
              </div>

              {/* Зона перетаскивания */}
              <Droppable droppableId={stage.id}>
                {(provided) => (
                  <div
                    ref={provided.innerRef}
                    {...provided.droppableProps}
                    class="space-y-3 min-h-[350px] flex-1"
                  >
                    {stageDeals.map((deal, index) => (
                      <Draggable key={deal.id} draggableId={deal.id} index={index}>
                        {(provided) => (
                          <div
                            ref={provided.innerRef}
                            {...provided.draggableProps}
                            {...provided.dragHandleProps}
                            class="p-4 rounded-lg bg-white dark:bg-slate-800 shadow-sm border border-slate-200 dark:border-slate-700 hover:border-indigo-500 dark:hover:border-indigo-400 transition-colors"
                          >
                            <h4 class="font-semibold text-xs mb-2 text-slate-750">{deal.title}</h4>
                            <div class="text-xs font-bold text-indigo-600 dark:text-indigo-400">
                              {deal.amount.toLocaleString("ru-RU")} ₽
                            </div>
                          </div>
                        )}
                      </Draggable>
                    ))}
                    {provided.placeholder}
                  </div>
                )}
              </Droppable>

              {/* Подвал колонки с итоговой суммой */}
              <div class="mt-4 pt-3 border-t border-slate-200 dark:border-slate-800 flex items-center justify-between text-xs font-bold text-slate-700 dark:text-slate-300">
                <span>Итого:</span>
                <span class="text-indigo-600 dark:text-indigo-400">{totalAmount.toLocaleString("ru-RU")} ₽</span>
              </div>
            </div>
          );
        })}
      </div>
    </DragDropContext>
  );
}

📹 2. RTSP/HLS Видеоплеер CameraPlayer.tsx

Для отображения видеопотока с камер безопасности в реальном времени мы используем библиотеку hls.js. Она преобразует фрагментированный HLS-стрим в понятный для HTML5 тега <video> медиа-поток на клиенте. 

"use client";

import React, { useEffect, useRef, useState } from "react";
import Hls from "hls.js";

interface CameraProps {
  streamUrl: string; // URL вида: http://localhost:3000/api/camera/stream.m3u8
  cameraName: string;
}

export default function CameraPlayer({ streamUrl, cameraName }: CameraProps) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const [error, setError] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    const video = videoRef.current;
    if (!video) return;

    let hls: Hls | null = null;

    // 1. Проверяем, поддерживает ли браузер воспроизведение HLS нативно (например, Safari)
    if (video.canPlayType("application/vnd.apple.mpegurl")) {
      video.src = streamUrl;
      video.addEventListener("loadedmetadata", () => {
        video.play();
        setIsLoading(false);
      });
      
      video.addEventListener("error", () => {
        setError("Ошибка воспроизведения видеопотока.");
        setIsLoading(false);
      });
    }
    // 2. Если не поддерживает (Chrome, Firefox), используем библиотеку hls.js
    else if (Hls.isSupported()) {
      hls = new Hls({
        maxBufferLength: 10, // Держим буфер не больше 10 секунд для минимизации задержки
        liveSyncDuration: 3,  // Синхронизируемся в 3 секундах от прямого эфира
        manifestLoadingMaxRetry: 5
      });

      hls.loadSource(streamUrl);
      hls.attachMedia(video);

      hls.on(Hls.Events.MANIFEST_PARSED, () => {
        video.play().catch(() => {
          // Игнорируем ошибку автоплея, если пользователь еще не кликнул на страницу
        });
        setIsLoading(false);
      });

      hls.on(Hls.Events.ERROR, (event, data) => {
        if (data.fatal) {
          switch (data.type) {
            case Hls.ErrorTypes.NETWORK_ERROR:
              setError("Сеть: Ошибка загрузки HLS файлов.");
              hls?.startLoad(); // Пробуем перезагрузить
              break;
            case Hls.ErrorTypes.MEDIA_ERROR:
              setError("Медиа: Ошибка декодирования кадров.");
              hls?.recoverMediaError(); // Пробуем восстановить плеер
              break;
            default:
              setError("Критическая ошибка воспроизведения.");
              hls?.destroy();
              break;
          }
        }
      });
    } else {
      setError("Ваш браузер не поддерживает HLS стриминг.");
      setIsLoading(false);
    }

    // Очистка ресурсов при размонтировании плеера (Важно!)
    return () => {
      if (hls) {
        hls.destroy();
      }
    };
  }, [streamUrl]);

  return (
    <div class="w-full bg-slate-950 rounded-xl overflow-hidden shadow-lg border border-slate-900 flex flex-col">
      {/* Шапка плеера */}
      <div class="px-4 py-2 bg-slate-900 border-b border-slate-800 flex items-center justify-between">
        <span class="text-xs font-semibold text-slate-300 flex items-center gap-1.5">
          <span class="h-2 w-2 rounded-full bg-emerald-500 animate-pulse"></span>
          {cameraName} — LIVE
        </span>
      </div>

      {/* Контейнер видео */}
      <div class="relative aspect-video flex items-center justify-center bg-black">
        {isLoading && !error && (
          <div class="absolute inset-0 flex flex-col items-center justify-center text-xs text-slate-400 gap-2 bg-slate-950">
            <svg class="animate-spin h-5 w-5 text-indigo-500" fill="none" viewBox="0 0 24 24">
              <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
              <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
            </svg>
            Подключение к IP-камере...
          </div>
        )}

        {error && (
          <div class="absolute inset-0 flex flex-col items-center justify-center text-xs text-rose-400 p-4 text-center bg-slate-950">
            ⚠️ {error}
          </div>
        )}

        <video
          ref={videoRef}
          muted
          playsInline
          class="w-full h-full object-cover"
        ></video>
      </div>
    </div>
  );
}
🚨 Критическая утечка памяти (Memory Leak): В шаге очистки хука обязательно вызывайте `hls.destroy()`. Если этого не сделать, при закрытии модального окна с камерой фоновые сетевые запросы hls.js продолжат скачивать видео-сегменты в фоновом режиме, что приведет к переполнению ОЗУ вкладки браузера через 10-15 минут и падению вкладки с ошибкой Out Of Memory.
