💻 Разбор кода модуля Whiteboard

Векторная графика в реальном времени требует от разработчика понимания работы с бесконечным холстом и WebSocket сокетами. Разберем ключевые файлы, отвечающие за интерактивность и реалтайм синхронизацию.

🖼 1. Бесконечный холст Canvas.tsx

Чтобы сделать холст бесконечным, мы не рендерим гигантскую SVG-картинку размером 100 000 на 100 000 пикселей. Вместо этого мы храним в React-стейтах три числа: panX (сдвиг по горизонтали), panY (сдвиг по вертикали) и scale (масштаб).

Все фигуры мы рендерим внутри одного контейнера <g> в SVG, к которому применяем CSS-трансформацию `transform: translate(panX, panY) scale(scale)`. 

"use client";

import React, { useRef, useState } from "react";
import { useWhiteboardSocket } from "@/hooks/useWhiteboardSocket";

interface Shape {
  id: string;
  type: "RECTANGLE" | "CIRCLE" | "STICKER";
  x: number;
  y: number;
  width: number;
  height: number;
  fill: string;
}

export default function Canvas({ whiteboardId }: { whiteboardId: string }) {
  const svgRef = useRef<SVGSVGElement>(null);
  
  // Координаты смещения холста (Панорамирование)
  const [pan, setPan] = useState({ x: 0, y: 0 });
  const [scale, setScale] = useState(1);
  const [isPanning, setIsPanning] = useState(false);
  const [startPan, setStartPan] = useState({ x: 0, y: 0 });

  // Локальные фигуры
  const [shapes, setShapes] = useState<Shape[]>([]);
  // Инструмент рисования: "SELECT" | "RECTANGLE" | "PAN"
  const [tool, setTool] = useState<"SELECT" | "RECTANGLE" | "PAN">("SELECT");

  // Подключаем WebSocket сокеты
  const { emitShapeUpdate, emitCursorMove, activeCursors } = useWhiteboardSocket({
    whiteboardId,
    onRemoteShapeUpdate: (updatedShape) => {
      setShapes(prev => {
        const filtered = prev.filter(s => s.id !== updatedShape.id);
        return [...filtered, updatedShape];
      });
    }
  });

  // Преобразование экранных координат мыши в координаты на бесконечном холсте
  const getCanvasCoords = (clientX: number, clientY: number) => {
    if (!svgRef.current) return { x: 0, y: 0 };
    const rect = svgRef.current.getBoundingClientRect();
    
    // Вычитаем позицию SVG-элемента на экране, затем учитываем сдвиг (pan) и масштаб (scale)
    const x = (clientX - rect.left - pan.x) / scale;
    const y = (clientY - rect.top - pan.y) / scale;
    return { x, y };
  };

  const handleMouseDown = (e: React.MouseEvent) => {
    if (tool === "PAN" || e.button === 1) { // Клик колесиком мыши или активный режим PAN
      setIsPanning(true);
      setStartPan({ x: e.clientX - pan.x, y: e.clientY - pan.y });
      return;
    }

    if (tool === "RECTANGLE") {
      const { x, y } = getCanvasCoords(e.clientX, e.clientY);
      const newShape: Shape = {
        id: crypto.randomUUID(),
        type: "RECTANGLE",
        x,
        y,
        width: 100,
        height: 100,
        fill: "#a5b4fc" // Нежно-синий
      };
      
      setShapes(prev => [...prev, newShape]);
      emitShapeUpdate(newShape); // Отправляем на бэк по веб-сокетам
      setTool("SELECT"); // Возвращаемся в режим выбора
    }
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    if (isPanning) {
      setPan({
        x: e.clientX - startPan.x,
        y: e.clientY - startPan.y
      });
      return;
    }

    // Транслируем положение курсора мыши другим пользователям
    const { x, y } = getCanvasCoords(e.clientX, e.clientY);
    emitCursorMove({ x, y });
  };

  const handleMouseUp = () => {
    setIsPanning(false);
  };

  // Зум колесиком мыши (с привязкой масштабирования к курсору мыши)
  const handleWheel = (e: React.WheelEvent) => {
    e.preventDefault();
    const zoomFactor = 1.1;
    const nextScale = e.deltaY < 0 ? scale * zoomFactor : scale / zoomFactor;
    
    // Ограничиваем зум от 10% до 500%
    const boundedScale = Math.min(Math.max(nextScale, 0.1), 5);
    setScale(boundedScale);
  };

  return (
    <div class="relative w-full h-[600px] bg-slate-50 dark:bg-slate-950 border border-slate-200 dark:border-slate-800 rounded-xl overflow-hidden">
      {/* Панель инструментов */}
      <div class="absolute top-4 left-4 z-10 flex gap-2 bg-white dark:bg-slate-900 p-2 rounded-lg shadow-md border dark:border-slate-800">
        <button onClick={() => setTool("SELECT")} class={`px-3 py-1.5 rounded text-xs ${tool === "SELECT" ? "bg-indigo-600 text-white" : "hover:bg-slate-100 dark:hover:bg-slate-800"}`}>Выбор</button>
        <button onClick={() => setTool("RECTANGLE")} class={`px-3 py-1.5 rounded text-xs ${tool === "RECTANGLE" ? "bg-indigo-600 text-white" : "hover:bg-slate-100 dark:hover:bg-slate-800"}`}>+ Прямоугольник</button>
        <button onClick={() => setTool("PAN")} class={`px-3 py-1.5 rounded text-xs ${tool === "PAN" ? "bg-indigo-600 text-white" : "hover:bg-slate-100 dark:hover:bg-slate-800"}`}>Рука (Pan)</button>
      </div>

      <svg
        ref={svgRef}
        onMouseDown={handleMouseDown}
        onMouseMove={handleMouseMove}
        onMouseUp={handleMouseUp}
        onWheel={handleWheel}
        class="w-full h-full cursor-grab active:cursor-grabbing outline-none"
      >
        {/* Контейнер с бесконечной матрицей сдвига и зума */}
        <g transform={`translate(${pan.x}, ${pan.y}) scale(${scale})`}>
          {/* Координатная сетка на фоне */}
          <defs>
            <pattern id="grid" width="40" height="40" patternUnits="userSpaceOnUse">
              <path d="M 40 0 L 0 0 0 40" fill="none" stroke="rgba(148, 163, 184, 0.1)" stroke-width="1" />
            </pattern>
          </defs>
          <rect width="100000" height="100000" x="-50000" y="-50000" fill="url(#grid)" pointer-events="none" />

          {/* Отрисовка фигур */}
          {shapes.map(shape => (
            <rect
              key={shape.id}
              x={shape.x}
              y={shape.y}
              width={shape.width}
              height={shape.height}
              fill={shape.fill}
              stroke="#4f46e5"
              stroke-width="2"
              rx="4"
            />
          ))}

          {/* Отрисовка чужих курсоров */}
          {Object.entries(activeCursors).map(([userId, cursor]: [string, any]) => (
            <g key={userId} transform={`translate(${cursor.x}, ${cursor.y})`}>
              <polygon points="0,0 4,14 7,10 12,12" fill={cursor.color || "#ef4444"} />
              <rect x="10" y="10" width="70" height="18" fill={cursor.color || "#ef4444"} rx="3" />
              <text x="14" y="22" fill="#fff" font-size="10" font-family="sans-serif" font-weight="bold">
                {cursor.name || "Гость"}
              </text>
            </g>
          ))}
        </g>
      </svg>
    </div>
  );
}

📡 2. Синхронизация по сокетам useWhiteboardSocket.ts

Этот хук отвечает за сетевое взаимодействие с сервером WebSocket на базе Socket.io. Он подписывается на комнату конкретной доски и транслирует локальные события в эфир. 

import { useEffect, useRef, useState } from "react";
import { io, Socket } from "socket.io-client";

interface UseSocketProps {
  whiteboardId: string;
  onRemoteShapeUpdate: (shape: any) => void;
}

export function useWhiteboardSocket({ whiteboardId, onRemoteShapeUpdate }: UseSocketProps) {
  const socketRef = useRef<Socket | null>(null);
  const [activeCursors, setActiveCursors] = useState<Record<string, any>>({});

  useEffect(() => {
    // Подключаемся к WebSocket серверу
    const socketUrl = process.env.NEXT_PUBLIC_WS_SERVER_URL || "http://localhost:3001";
    const socket = io(socketUrl, {
      transports: ["websocket"] // Принудительно используем быстрые вебсокеты без HTTP-polling
    });
    
    socketRef.current = socket;

    // Входим в комнату нашей доски
    socket.emit("whiteboard:join", { whiteboardId });

    // Слушаем создание и обновление фигур другими разработчиками
    socket.on("shape:updated", (shape: any) => {
      onRemoteShapeUpdate(shape);
    });

    // Слушаем перемещение курсоров других пользователей
    socket.on("cursor:moved", (data: { userId: string; name: string; x: number; y: number; color: string }) => {
      setActiveCursors(prev => ({
        ...prev,
        [data.userId]: { x: data.x, y: data.y, name: data.name, color: data.color }
      }));
    });

    // Очистка при выходе из проекта или закрытии вкладки
    socket.on("user:left", ({ userId }) => {
      setActiveCursors(prev => {
        const copy = { ...prev };
        delete copy[userId];
        return copy;
      });
    });

    return () => {
      socket.disconnect();
    };
  }, [whiteboardId]);

  // Метод отправки изменений фигуры в WebSocket-канал
  const emitShapeUpdate = (shape: any) => {
    if (socketRef.current) {
      socketRef.current.emit("shape:update", { whiteboardId, shape });
    }
  };

  // Метод отправки координат курсора мыши (срабатывает на mouseMove)
  const emitCursorMove = (coords: { x: number; y: number }) => {
    if (socketRef.current) {
      // Отправляем сырые координаты без тяжелого логгирования
      socketRef.current.emit("cursor:move", { whiteboardId, ...coords });
    }
  };

  return {
    emitShapeUpdate,
    emitCursorMove,
    activeCursors
  };
}
⚠️ Важная деталь производительности: Метод `emitCursorMove` вызывается при каждом срабатывании события `MouseMove` в браузере (до 60-120 раз в секунду). Чтобы не спамить сетевой канал, на бэкенде обязательно применяется дросселирование (Throttling) отправки броадкастов на стороне сервера WebSocket (сервер пересылает курсоры не чаще, чем раз в 30 миллисекунд).
